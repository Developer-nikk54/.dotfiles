# keybinds
search = s
find = / or ?
select all = S -> b
delete mode options = d
goto mode options = g
fold mode options = z
change mode options = c
visual mode options = v
indent (left/right) = < >
window mode options = C-w
mark mode = '
git = <leader>gg
restore last session = <leader>qs / <leader>ql
open floating terminal = C-_  --> toggle terminal C-/   --note: only work in neovide or tmux

# commands
command history = <leader> :
paste yank text in command mode using registers = C-r -> +
set nospell/spell = spell check onn/off
set foldmethod=manual = manual code fold using visual mode keymap:zf/za
set foldmethod=expr = automatic code fold keymap:za

text manipulation
------------------------
select specific part = C- <leader>

scroll
-------------
line-by-line = C-e / C-y
forward/backward (full page) = C-f / C-b
down/up = C-d / C-u


toggle settings
--------------------
code formatter (prettier) = <leader>uf
create fold = zf / za


find and searchs
--------------------
find file = <leader>ff
find buffer all = <leader>, / <leader>fB
find project = <leader> p



plugins specific
--------------------
yanky.nvim = {
  p = paste
  C-p = next text paste
  C-n = previouse text paste
}
